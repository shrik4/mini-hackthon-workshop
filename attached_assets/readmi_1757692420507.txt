# 1) MVP scope (what to ship)

* Input: short problem statement (e.g., “Smart Water Bottle”).
* Output (single click / POST):

  1. Market research summary (top competitors + 3 bullets).
  2. Suggested 5 features prioritized (MVP vs stretch).
  3. Small React demo scaffold (frontend + simple FastAPI backend) packaged as a ZIP.
  4. 3-slide pitch PDF (problem, solution+features, ask/next steps).
* Bonus: live demo URL (ngrok) or local run + recording.

# 2) Tech stack (fast & hackable)

* Backend: FastAPI (Python)
* LLM: OpenAI (or any GPT provider) via their Python SDK
* Web search (competitor research): SerpAPI / Bing Web Search (quick integration)
* Templating / code generation: Jinja2 (render code files), zipfile for packaging
* Pitch deck: ReportLab (create simple 3-page PDF)
* Frontend: React (Vite) — but scaffold created by templates
* Dev tools: uvicorn, ngrok (for live demo)
* Storage: in-memory (no DB) for one-day MVP

### b) orchestrator (backend/generator.py)

```python
# backend/generator.py
import os, io, zipfile, base64, asyncio
from jinja2 import Environment, FileSystemLoader
from reportlab.lib.pagesizes import letter
from reportlab.pdfgen import canvas
from openai import OpenAI
from serpapi import GoogleSearch  # optional; you can swap search provider

OPENAI_KEY = os.getenv("OPENAI_API_KEY")
SERPAPI_KEY = os.getenv("SERPAPI_KEY")
client = OpenAI(api_key=OPENAI_KEY)

env = Environment(loader=FileSystemLoader("templates"))

async def call_llm(prompt, temperature=0.2):
    resp = client.chat.completions.create(model="gpt-4o-mini", messages=[{"role":"user","content":prompt}], temperature=temperature)
    return resp.choices[0].message["content"]

def web_search(query, num=5):
    params = {"q": query, "engine":"google", "api_key": SERPAPI_KEY, "num": num}
    search = GoogleSearch(params)
    results = search.get_dict()
    snippets = []
    for r in results.get("organic_results", [])[:num]:
        snippets.append({"title": r.get("title"), "link": r.get("link"), "snippet": r.get("snippet")})
    return snippets

def create_pdf(title, problem, solution, features):
    buf = io.BytesIO()
    c = canvas.Canvas(buf, pagesize=letter)
    w, h = letter
    # Slide 1: Title / Problem
    c.setFont("Helvetica-Bold", 28); c.drawCentredString(w/2, h-100, title)
    c.setFont("Helvetica", 12); c.drawString(80, h-160, "Problem:")
    c.setFont("Helvetica", 11); c.drawString(80, h-180, problem)
    c.showPage()
    # Slide 2: Solution + Features
    c.setFont("Helvetica-Bold", 20); c.drawString(80, h-100, "Solution & Features")
    y = h-140
    for f in features:
        c.setFont("Helvetica-Bold", 12); c.drawString(90, y, "• " + f.get("title", f))
        y -= 18
    c.showPage()
    # Slide 3: Ask / Next steps
    c.setFont("Helvetica-Bold", 20); c.drawString(80, h-100, "Demo / Ask")
    c.setFont("Helvetica", 12); c.drawString(80, h-140, "Try the demo and see the generated app. Ask: Early testers / mentors / next steps.")
    c.showPage()
    c.save()
    buf.seek(0)
    return buf.read()

def make_react_zip(context):
    # context contains replacements like {project_name, features}
    buf = io.BytesIO()
    z = zipfile.ZipFile(buf, "w", zipfile.ZIP_DEFLATED)
    # Render templates
    for tpl_name in ["package.json.j2","src/App.jsx.j2","src/main.jsx.j2","index.html.j2"]:
        tpl = env.get_template(tpl_name)
        content = tpl.render(**context)
        z.writestr(tpl_name.replace(".j2",""), content)
    z.close()
    buf.seek(0)
    return buf.read()

async def orchestrate_generation(query):
    # 1) quick web search
    search_hits = web_search(query)
    # 2) LLM market summary (feed search results as part of prompt)
    search_text = "\n".join([f"{s['title']} - {s['link']}" for s in search_hits])
    market_prompt = f"You're HackPal. Given the problem: '{query}', summarize competitor landscape from these results:\n{search_text}\nGive 5-sentence summary and 3 competitors as bullets."
    market = await asyncio.to_thread(call_llm, market_prompt)
    # 3) Feature suggestion
    feat_prompt = f"Given the problem: '{query}' and market summary: {market}, list 5 prioritized features (MVP first) with 1-line dev note each."
    features_text = await asyncio.to_thread(call_llm, feat_prompt)
    # Parse features into a list (simple split)
    features = [{"title": line.strip().lstrip("-•0123456789. ").strip()} for line in features_text.splitlines() if line.strip()]
    # 4) Generate React + API scaffold
    context = {"project_name": query.replace(" ", "-").lower(), "features": features}
    zip_bytes = make_react_zip(context)
    # 5) Create 3-page pitch PDF
    pdf_bytes = create_pdf(title=f"HackPal Demo - {query}", problem=query, solution="Auto-generated demo app + pitch", features=features)
    return {
        "market": market,
        "features": features_text,
        "zip_bytes": zip_bytes,
        "pdf_bytes": pdf_bytes,
        "zip_base64": base64.b64encode(zip_bytes).decode(),
        "pdf_base64": base64.b64encode(pdf_bytes).decode()
    }
```

### c) Jinja templates (one example)

```jsx
import React, {useState} from "react";

export default function App(){
  const [res, setRes] = useState(null);
  const [q, setQ] = useState("{{ project_name }}");
  async function go(){
    const fd = new FormData();
    fd.append("query", q);
    const r = await fetch("/api/generate", {method: "POST", body: fd});
    const data = await r.json();
    setRes(data);
  }
  return (
    <div style={{padding:20}}>
      <h1>HackPal — Demo</h1>
      <input value={q} onChange={e=>setQ(e.target.value)} />
      <button onClick={go}>Generate</button>
      {res && <pre>{JSON.stringify(res, null, 2)}</pre>}
    </div>
  );
}
```

# 10) Pitfalls & quick fixes

* **API rate limits / cost**: use smaller model + caching for search results. Pre-generate fallback outputs.
* **Search reliability**: SerpAPI is simplest in a hackathon; you can fallback to LLM-only summarization if no search key.
* **PDF conversion**: ReportLab creates native PDFs — simple and reliable. If you prefer slides (.pptx), use `python-pptx` but conversion to PDF needs LibreOffice/soffice.

import React, { useState } from "react";

// Simple typewriter effect
const Typewriter = ({ text }) => {
  const [display, setDisplay] = useState("");
  React.useEffect(() => {
    let i = 0;
    const interval = setInterval(() => {
      setDisplay(text.slice(0, i));
      i++;
      if (i > text.length) clearInterval(interval);
    }, 100);
    return () => clearInterval(interval);
  }, [text]);
  return <span>{display}</span>;
};

export default function App() {
  const [query, setQuery] = useState("");
  const [loading, setLoading] = useState(false);
  const [result, setResult] = useState(null);
  const [activeTab, setActiveTab] = useState("market");

  async function handleGenerate() {
    setLoading(true);
    setResult(null);
    const fd = new FormData();
    fd.append("query", query);
    try {
      const res = await fetch("http://localhost:8000/api/generate", {
        method: "POST",
        body: fd,
      });
      const data = await res.json();
      setResult(data);
    } catch (err) {
      console.error(err);
    } finally {
      setLoading(false);
    }
  }

  const downloadFile = (base64, filename, type) => {
    const link = document.createElement("a");
    link.href = `data:${type};base64,${base64}`;
    link.download = filename;
    link.click();
  };

  return (
    <div className="min-h-screen bg-[#0f172a] text-white flex flex-col items-center p-8">
      <h1 className="text-4xl font-bold text-transparent bg-clip-text bg-gradient-to-r from-pink-500 to-violet-500 mb-6">
        <Typewriter text="🚀 HackPal: Your Hackathon Teammate" />
      </h1>

      <div className="w-full max-w-2xl bg-gradient-to-r from-indigo-500/20 to-purple-500/20 rounded-2xl p-6 shadow-xl">
        <textarea
          className="w-full p-4 rounded-lg bg-black/30 border border-purple-500/30 focus:ring-2 focus:ring-purple-500 outline-none"
          rows={3}
          placeholder="Enter your hackathon idea (e.g. Smart Water Bottle)..."
          value={query}
          onChange={(e) => setQuery(e.target.value)}
        />
        <button
          onClick={handleGenerate}
          disabled={loading || !query}
          className="mt-4 px-6 py-3 rounded-xl bg-gradient-to-r from-pink-600 to-purple-600 hover:from-pink-500 hover:to-purple-500 transition-all w-full"
        >
          {loading ? "Generating..." : "Generate Project 🚀"}
        </button>
      </div>

      {loading && (
        <div className="mt-8 animate-pulse text-lg text-purple-400">
          ⏳ Crunching ideas, writing code, and making slides...
        </div>
      )}

      {result && (
        <div className="w-full max-w-3xl mt-10">
          {/* Tab navigation */}
          <div className="flex gap-4 border-b border-purple-500/40 mb-4">
            {["market", "features", "pitch", "download"].map((tab) => (
              <button
                key={tab}
                className={`pb-2 capitalize ${
                  activeTab === tab
                    ? "border-b-2 border-pink-500 text-pink-400"
                    : "text-gray-400 hover:text-white"
                }`}
                onClick={() => setActiveTab(tab)}
              >
                {tab}
              </button>
            ))}
          </div>

          {/* Tab content */}
          {activeTab === "market" && (
            <pre className="bg-black/40 p-4 rounded-xl whitespace-pre-wrap">
              {result.market}
            </pre>
          )}
          {activeTab === "features" && (
            <pre className="bg-black/40 p-4 rounded-xl whitespace-pre-wrap">
              {result.features}
            </pre>
          )}
          {activeTab === "pitch" && (
            <div className="bg-black/40 p-6 rounded-xl">
              <p className="mb-2">📄 Pitch Deck Generated!</p>
              <button
                onClick={() =>
                  downloadFile(result.pdf_base64, "HackPal-Pitch.pdf", "application/pdf")
                }
                className="px-4 py-2 rounded-lg bg-purple-600 hover:bg-purple-500"
              >
                Download Pitch Deck
              </button>
            </div>
          )}
          {activeTab === "download" && (
            <div className="bg-black/40 p-6 rounded-xl">
              <p className="mb-2">💻 Project Scaffold Ready!</p>
              <button
                onClick={() =>
                  downloadFile(result.zip_base64, "HackPal-App.zip", "application/zip")
                }
                className="px-4 py-2 rounded-lg bg-pink-600 hover:bg-pink-500"
              >
                Download React App
              </button>
            </div>
          )}
        </div>
      )}
    </div>
  );
}
